#
# The function accepts following parameters:
#  1. INTEGER s
#  2. INTEGER t
#  3. INTEGER a
#  4. INTEGER b
#  5. INTEGER_ARRAY apples
#  6. INTEGER_ARRAY oranges
#

def countApplesAndOranges(s, t, a, b, apples, oranges):
    m=len(apples)
    n=len(oranges)
    applecount=0
    orangecount=0
    for i in range(max(m,n)):
        if i<m:
            if s<= (apples[i]+a) <= t:
                applecount+=1
        if i<n:
            if s<= (oranges[i]+b)<=t:
                orangecount+=1
    print(applecount)
    print(orangecount)
