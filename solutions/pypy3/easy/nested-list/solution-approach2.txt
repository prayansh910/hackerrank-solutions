if __name__ == '__main__':
    records=[]
    for _ in range(int(input())):
        name = input()
        score = float(input())
        
        records.append([name,score])
    
    second_lowest=sorted(set(pair[1] for pair in records))
             
    names=sorted([pair[0] for pair in records if pair[1]==second_lowest[1]])
    for name in names:
        print(name)
