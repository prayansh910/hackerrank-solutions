# Insight     The algorithm maintains the current minimum and maximum scores observed so far, incrementing the respective counters only when a score strictly exceeds or falls below these established bounds.
# Interview   Before: "I would sort the array to find the min and max." After: "Sorting is unnecessary; a single O(n) linear scan is sufficient to track the records as they are broken, which is optimal given the input size constraints."
# Pitfalls    (1) Initializing min and max to zero instead of the first game score, which fails if all scores are negative.  (2) Using non-strict inequality (>= or <=) instead of strict inequality, which incorrectly counts games where the score merely matches the existing record.  (3) Failing to handle the first game correctly, as the problem states the first game establishes the initial record and cannot break it.

import math

import sys

#
# Complete the 'breakingRecords' function below.
#
# The function is expected to return an INTEGER_ARRAY.
# The function accepts INTEGER_ARRAY scores as parameter.
#

def breakingRecords(scores):
    min=scores[0]
    max=scores[0]
    record=[0,0]
    for i in scores:
        if i>max:
            max=i
            record[0]+=1
        elif i<min:
            min=i
            record[1]+=1
    return record

            

if __name__ == '__main__':
    fptr = open(os.environ['OUTPUT_PATH'], 'w')

    n = int(input().strip())

    scores = list(map(int, input().rstrip().split()))

    result = breakingRecords(scores)

    fptr.write(' '.join(map(str, result)))
    fptr.write('\n')

    fptr.close()
