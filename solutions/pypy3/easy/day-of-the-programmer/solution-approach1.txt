def dayOfProgrammer(year):
    yearl=""
    if year==1918:
        return "26.09.1918"
    if year<1918:
        if year%4==0:
            yearl="leap"
    else:
        if (year%4==0 and year%100!=0) or (year%400==0):
            yearl="leap"
    if yearl=="leap":
        dd="12"
    else:
        dd="13"
    return(f"{dd}.09.{year}")
