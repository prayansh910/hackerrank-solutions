# Insight     The solution identifies valid integers by checking multiples of the least common multiple of array a that are also divisors of the greatest common divisor of array b.
# Interview   Before: "I could iterate through all numbers from 1 to 100 and check every element in both arrays." After: "That is inefficient. Instead, I calculate the LCM of array a and the GCD of array b, then scan the range between them in O(N+M+K) time, where K is the range size."
# Pitfalls    (1) Failing to account for the range upper bound being inclusive, which requires range(amax, bmin + 1).  (2) Assuming the range is empty if the LCM of a exceeds the GCD of b, which the loop handles correctly by not executing.  (3) Neglecting that the integer must be a multiple of the LCM of a and a divisor of the GCD of b.

#!/bin/python3

import math
import os
import random
import re
import sys

#
# Complete the 'getTotalX' function below.
#
# The function is expected to return an INTEGER.
# The function accepts following parameters:
#  1. INTEGER_ARRAY a
#  2. INTEGER_ARRAY b
#

def getTotalX(a, b):
    amax=math.lcm(*a)
    bmin=math.gcd(*b)
    count=0
    for i in range(amax,bmin+1):
        if i%amax==0 and bmin%i==0:
            count+=1
    return count      

if __name__ == '__main__':
    fptr = open(os.environ['OUTPUT_PATH'], 'w')

    first_multiple_input = input().rstrip().split()

    arr = list(map(int, input().rstrip().split()))

    brr = list(map(int, input().rstrip().split()))

    total = getTotalX(arr, brr)

    fptr.write(str(total) + '\n')

    fptr.close()
