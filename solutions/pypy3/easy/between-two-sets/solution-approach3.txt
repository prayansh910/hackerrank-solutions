#
# Complete the 'getTotalX' function below.
#
# The function is expected to return an INTEGER.
# The function accepts following parameters:
#  1. INTEGER_ARRAY a
#  2. INTEGER_ARRAY b
#

def getTotalX(a, b):
    amax=math.lcm(*a)
    bmin=math.gcd(*b)
    count=0
    for i in range(amax,bmin+1,amax):
        if bmin%i==0:
            count+=1
    return count
