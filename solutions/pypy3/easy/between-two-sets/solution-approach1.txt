# Insight     The algorithm identifies integers that are multiples of the least common multiple of array a and divisors of the greatest common divisor of array b.
# Interview   Before: "I would iterate through all numbers between the arrays and check divisibility for every element." After: "By calculating the LCM of array a and the GCD of array b, I reduce the search space to multiples of the LCM, achieving O(n + m + (bmin/amax)) time complexity."
# Pitfalls    (1) Failing to account for the range constraint where the candidate integer must be a multiple of the LCM of array a.  (2) Assuming the candidate integer must be within the range of the minimum of array a and the maximum of array b instead of the LCM and GCD bounds.

#!/bin/python3

import math
import os
import random
import re
import sys

#
# Complete the 'getTotalX' function below.
#
# The function is expected to return an INTEGER.
# The function accepts following parameters:
#  1. INTEGER_ARRAY a
#  2. INTEGER_ARRAY b
#

def getTotalX(a, b):
    amax=math.lcm(*a)
    bmin=math.gcd(*b)
    result=[]
    for i in range(amax,bmin+1):
        if i%amax==0 and bmin%i==0:
            result.append(i)
    return len(result)      

if __name__ == '__main__':
    fptr = open(os.environ['OUTPUT_PATH'], 'w')

    first_multiple_input = input().rstrip().split()

    n = int(first_multiple_input[0])

    m = int(first_multiple_input[1])

    arr = list(map(int, input().rstrip().split()))

    brr = list(map(int, input().rstrip().split()))

    total = getTotalX(arr, brr)

    fptr.write(str(total) + '\n')

    fptr.close()
