# Insight     The code leverages the property that list.index returns the first occurrence of the maximum frequency, which naturally satisfies the requirement to return the smallest bird type ID in case of ties.
# Interview   Before: "I could sort the array and count frequencies." After: "Sorting would take O(n log n), but using a frequency array allows us to solve this in O(n) time and O(m) space, where m is the maximum bird ID, by finding the first index with the maximum count."
# Pitfalls    (1) The code assumes bird IDs are non-negative integers, which is consistent with the problem constraints.  (2) Using max(arr) to size the frequency array is efficient here, but could cause memory issues if bird IDs were extremely large sparse values.

#!/bin/python3

import math
import os
import random
import re
import sys

#
# Complete the 'migratoryBirds' function below.
#
# The function is expected to return an INTEGER.
# The function accepts INTEGER_ARRAY arr as parameter.
#

def migratoryBirds(arr):
    count=[0]*(max(arr)+1)
    for bird_type in arr:
        count[bird_type]+=1
    max_index=max(count)
    return count.index(max_index) #returns first occurance so we got lucky with this built-in function
    

if __name__ == '__main__':
    fptr = open(os.environ['OUTPUT_PATH'], 'w')

    arr_count = int(input().strip())

    arr = list(map(int, input().rstrip().split()))

    result = migratoryBirds(arr)

    fptr.write(str(result) + '\n')

    fptr.close()
