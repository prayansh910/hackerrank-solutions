# Insight     The algorithm iterates through the grades and applies the rounding rule only if the grade is at least 38 and the difference to the next multiple of 5 is strictly less than 3.
# Interview   Before: "How would you handle the rounding logic for each student?" After: "I iterate through the array in O(n) time, checking if the grade is at least 38 and if the remainder relative to 5 is less than 3, which ensures we only round up when the difference is 1 or 2."
# Pitfalls    (1) Failing to check the grade threshold of 38, which causes grades below 40 to be rounded incorrectly.  (2) Rounding grades where the difference to the next multiple of 5 is exactly 3, violating the rule that the difference must be less than 3.

#!/bin/python3

import math
import os
import random
import re
import sys

#
# Complete the 'gradingStudents' function below.
#
# The function is expected to return an INTEGER_ARRAY.
# The function accepts INTEGER_ARRAY grades as parameter.
#

def gradingStudents(grades):
    for i in range(len(grades)):
        if grades[i]>=38:
            remainder=5-(grades[i]%5)
            if remainder<3:
                grades[i]+=remainder
            
    return grades

if __name__ == '__main__':
    fptr = open(os.environ['OUTPUT_PATH'], 'w')

    grades_count = int(input().strip())

    grades = []

    for _ in range(grades_count):
        grades_item = int(input().strip())
        grades.append(grades_item)

    result = gradingStudents(grades)

    fptr.write('\n'.join(map(str, result)))
    fptr.write('\n')

    fptr.close()
